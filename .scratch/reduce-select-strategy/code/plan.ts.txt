/**
 * The select strategy of reduce. Each slice of the book proposes narrow
 * candidates with a chapter on every point, code chooses among them under the
 * budget, and one more call puts the choice in order. Generation keeps stations
 * faithful; selection keeps them spread. Why: docs/EVAL.md, "Open problem".
 */
import { type LlmProvider, parseJsonOutput } from '../llm/types';
import { CairnError } from '../errors';
import type { BookType, ChapterNote, NodeKind, PathNode, Stage } from '../types';
import { clampNodeMinutes, type ReadingBudget } from './budget';
import { memoryStore, runJob } from './job';
import type { Prompts } from './prompts';
import { type Candidate, selectCandidates, sourcesOf } from './select';
import { digestOf, repairStages, stageCount, type StageIds } from './stages';

/** Few enough chapters that none sits in a long context's neglected middle. */
const PART_SIZE = 12;
/** Enough candidates for selection to have a real choice, few enough for one call per slice. */
const CANDIDATES_PER_STATION = 2.5;

const KINDS: readonly NodeKind[] = ['concept', 'argument', 'event', 'character'];

const CANDIDATE_SCHEMA = {
  type: 'object', additionalProperties: false, required: ['candidates'],
  properties: {
    candidates: {
      type: 'array',
      items: {
        type: 'object', additionalProperties: false,
        required: ['title', 'kind', 'brief', 'keyPoints', 'estMinutes', 'weight'],
        properties: {
          title: { type: 'string' },
          kind: { type: 'string', enum: KINDS as unknown as string[] },
          brief: { type: 'string' },
          keyPoints: {
            type: 'array',
            items: {
              type: 'object', additionalProperties: false, required: ['text', 'chapter'],
              properties: { text: { type: 'string' }, chapter: { type: 'integer' } },
            },
          },
          estMinutes: { type: 'number' },
          weight: { type: 'integer', enum: [1, 2, 3] },
        },
      },
    },
  },
} as const;

const STAGES_PROPERTY = {
  type: 'array',
  items: {
    type: 'object', additionalProperties: false, required: ['title', 'ids'],
    properties: { title: { type: 'string' }, ids: { type: 'array', items: { type: 'string' } } },
  },
} as const;

const ARRANGE_SCHEMA = {
  type: 'object', additionalProperties: false, required: ['stages'], properties: { stages: STAGES_PROPERTY },
} as const;

const REFINE_SCHEMA = {
  type: 'object', additionalProperties: false, required: ['problems', 'drop', 'stages'],
  properties: {
    problems: { type: 'array', items: { type: 'string' } },
    drop: { type: 'array', items: { type: 'string' } },
    stages: STAGES_PROPERTY,
  },
} as const;

export interface PlanRequest {
  readonly notes: readonly ChapterNote[];
  readonly type: BookType;
  readonly budget: ReadingBudget;
  readonly targetNodes: number;
  readonly prompts: Prompts;
  readonly budgetLabel: string;
  readonly coverage: string;
  readonly refine: boolean;
  readonly signal?: AbortSignal;
}

export interface PlanResult {
  readonly nodes: readonly PathNode[];
  readonly stages: readonly Stage[];
  /** Candidates rejected for citing no chapter of their own slice. */
  readonly dropped: number;
}

export async function planPath(request: PlanRequest, provider: LlmProvider): Promise<PlanResult> {
  const { candidates, dropped } = await propose(request, provider);
  const chosen = selectCandidates(candidates, request.budget);
  if (chosen.length === 0) throw new CairnError('reduce_empty');

  const ids = chosen.map((_, i) => `s${i}`);
  const byId = new Map(chosen.map((c, i) => [`s${i}`, c]));
  const arranged = await arrange(request, provider, ids, byId);
  const final = request.refine ? await refine(request, provider, arranged, byId) : arranged;
  return { ...toPath(final, byId, request.budget), dropped };
}

async function propose(request: PlanRequest, provider: LlmProvider): Promise<Proposed> {
  const { notes, prompts, budget } = request;
  const parts = slices(notes);
  const outline = notes.map((n) => `[${n.idx}] ${n.title}`).join('\n');
  const tasks = parts.map((part, i) => ({ id: `part${i}`, input: { part, i } }));

  const job = await runJob(tasks, async ({ part, i }) => {
    const first = part[0]?.idx ?? 0;
    const last = part.at(-1)?.idx ?? first;
    const count = Math.max(2, Math.ceil(CANDIDATES_PER_STATION * request.targetNodes * part.length / notes.length));
    const raw = await provider.complete({
      label: `reduce.propose:${first}-${last}`,
      system: prompts.plan.propose.system(request.type, request.coverage),
      prompt: prompts.plan.propose.user({
        outline, part: digestOf(part, Infinity), firstChapter: first, lastChapter: last,
        budgetLabel: request.budgetLabel, count, minutesPerNode: budget.minutesPerNode,
      }),
      schema: CANDIDATE_SCHEMA,
      signal: request.signal,
    });
    return normalize(parseJsonOutput<{ candidates?: unknown }>(raw).candidates, part, i, budget);
  }, memoryStore<Proposed>(), { concurrency: provider.suggestedConcurrency, signal: request.signal });

  const results = [...job.results.values()];
  // Every slice failing is a failed reduce; some failing is a thinner choice, and the path still covers the rest
  if (results.length === 0) throw job.failures[0]?.error ?? new CairnError('reduce_empty');
  return {
    candidates: results.flatMap((r) => r.candidates),
    dropped: results.reduce((sum, r) => sum + r.dropped, 0),
  };
}

interface Proposed {
  readonly candidates: readonly Candidate[];
  readonly dropped: number;
}

/** Balanced slices of at most PART_SIZE chapters, in book order. */
export function slices<T>(notes: readonly T[]): readonly (readonly T[])[] {
  const count = Math.max(1, Math.ceil(notes.length / PART_SIZE));
  const size = Math.ceil(notes.length / count);
  return Array.from({ length: count }, (_, i) => notes.slice(i * size, (i + 1) * size)).filter((s) => s.length > 0);
}

/** A point may only cite a chapter of its own slice: the others were seen as titles, never read. */
export function normalize(
  raw: unknown, part: readonly ChapterNote[], partIndex: number, budget: ReadingBudget,
): Proposed {
  const own = new Set(part.map((n) => n.idx));
  const items = Array.isArray(raw) ? raw.filter((x): x is Record<string, unknown> => typeof x === 'object' && x !== null) : [];
  const candidates = items.flatMap((item): Candidate[] => {
    const points = (Array.isArray(item.keyPoints) ? item.keyPoints : []).flatMap((p) => {
      const point = p as { text?: unknown; chapter?: unknown };
      const text = typeof point.text === 'string' ? point.text.trim() : '';
      const chapter = Number(point.chapter);
      return text && own.has(chapter) ? [{ text, chapter }] : [];
    });
    const title = typeof item.title === 'string' ? item.title.trim() : '';
    if (!title || points.length === 0) return [];
    const weight = Number(item.weight);
    return [{
      title,
      kind: KINDS.includes(item.kind as NodeKind) ? (item.kind as NodeKind) : 'concept',
      brief: typeof item.brief === 'string' ? item.brief.trim() : '',
      points,
      estMinutes: clampNodeMinutes(Number(item.estMinutes), budget),
      weight: [1, 2, 3].includes(weight) ? weight : 1,
      part: partIndex,
    }];
  });
  return { candidates, dropped: items.length - candidates.length };
}

async function arrange(
  request: PlanRequest, provider: LlmProvider, ids: readonly string[], byId: ReadonlyMap<string, Candidate>,
): Promise<readonly StageIds[]> {
  const { prompts } = request;
  const raw = await provider.complete({
    label: 'reduce.arrange',
    system: prompts.plan.arrange.system(request.type),
    prompt: prompts.plan.arrange.user({ stations: listing(ids, byId), stageCount: stageCount(ids.length) }),
    schema: ARRANGE_SCHEMA,
    signal: request.signal,
  });
  const parsed = parseJsonOutput<{ stages?: unknown }>(raw);
  return repairStages(stagesIn(parsed.stages), ids, prompts.reduce.fallbackStage);
}

async function refine(
  request: PlanRequest, provider: LlmProvider, arranged: readonly StageIds[], byId: ReadonlyMap<string, Candidate>,
): Promise<readonly StageIds[]> {
  const { prompts } = request;
  const ids = arranged.flatMap((s) => s.ids);
  const path = arranged.map((s) => `${s.title}\n${listing(s.ids, byId)}`).join('\n\n');
  const raw = await provider.complete({
    label: 'reduce.refine',
    system: prompts.plan.refine.system,
    prompt: prompts.plan.refine.user({ path, digest: digestOf(request.notes), stageCount: stageCount(ids.length) }),
    schema: REFINE_SCHEMA,
    signal: request.signal,
  });
  const parsed = parseJsonOutput<{ drop?: unknown; stages?: unknown }>(raw);
  const drop = new Set(Array.isArray(parsed.drop) ? parsed.drop.filter((d): d is string => typeof d === 'string') : []);
  const kept = ids.filter((id) => !drop.has(id));
  // A review that would drop every station has misread its job; keep the arranged path
  if (kept.length === 0) return arranged;
  return repairStages(stagesIn(parsed.stages), kept, prompts.reduce.fallbackStage);
}

function toPath(
  stages: readonly StageIds[], byId: ReadonlyMap<string, Candidate>, budget: ReadingBudget,
): { nodes: readonly PathNode[]; stages: readonly Stage[] } {
  const order = stages.flatMap((s) => s.ids);
  const nodeId = new Map(order.map((id, i) => [id, `n${i}`]));
  const nodes = order.flatMap((id, idx): PathNode[] => {
    const c = byId.get(id);
    if (!c) return [];
    return [{
      id: `n${idx}`, idx, title: c.title, kind: c.kind, brief: c.brief,
      keyPoints: c.points.map((p) => p.text),
      sourceChapters: sourcesOf(c),
      estMinutes: clampNodeMinutes(c.estMinutes, budget),
    }];
  });
  return {
    nodes,
    stages: stages.map((s) => ({ title: s.title, nodeIds: s.ids.flatMap((id) => nodeId.get(id) ?? []) })),
  };
}

const listing = (ids: readonly string[], byId: ReadonlyMap<string, Candidate>): string =>
  ids.flatMap((id) => {
    const c = byId.get(id);
    return c ? [`[${id}] ${c.title} (chapters ${sourcesOf(c).join(',')}) — ${c.brief}`] : [];
  }).join('\n');

const stagesIn = (v: unknown): readonly { title?: unknown; ids?: unknown }[] =>
  Array.isArray(v) ? v.filter((x): x is Record<string, unknown> => typeof x === 'object' && x !== null) : [];
