/** Grouping a path's stations into stages, and repairing a model's grouping into a valid one. */
import type { ChapterNote } from '../types';

export interface StageIds {
  readonly title: string;
  readonly ids: readonly string[];
}

/** Stages must group, so keep at least two stations in each; one-per-stage is no grouping. */
export function stageCount(nodeCount: number): number {
  return Math.min(5, Math.max(1, Math.round(nodeCount / 4)));
}

/**
 * Every id exactly once, in the model's order where it gave one. An id it
 * dropped still has to be walked, so it joins the last stage rather than vanish.
 */
export function repairStages(
  raw: readonly { readonly title?: unknown; readonly ids?: unknown }[],
  ids: readonly string[],
  fallback: (n: number) => string,
): readonly StageIds[] {
  const known = new Set(ids);
  const seen = new Set<string>();
  const stages = raw.flatMap((stage, n) => {
    const kept: string[] = [];
    for (const id of Array.isArray(stage.ids) ? stage.ids : []) {
      if (typeof id !== 'string' || !known.has(id) || seen.has(id)) continue;
      seen.add(id);
      kept.push(id);
    }
    const title = typeof stage.title === 'string' && stage.title.trim() ? stage.title.trim() : fallback(n + 1);
    return kept.length > 0 ? [{ title, ids: kept }] : [];
  });

  const missing = ids.filter((id) => !seen.has(id));
  const last = stages.at(-1);
  const complete = missing.length === 0 ? stages
    : last ? [...stages.slice(0, -1), { ...last, ids: [...last.ids, ...missing] }]
    : [{ title: fallback(1), ids: missing }];
  return mergeSingles(complete);
}

function mergeSingles(stages: readonly StageIds[]): readonly StageIds[] {
  if (stages.length < 2) return stages;
  return stages.reduce<StageIds[]>((out, stage) => {
    const previous = out.at(-1);
    if (stage.ids.length < 2 && previous) {
      return [...out.slice(0, -1), { ...previous, ids: [...previous.ids, ...stage.ids] }];
    }
    if (previous && previous.ids.length < 2) {
      return [...out.slice(0, -1), { title: stage.title, ids: [...previous.ids, ...stage.ids] }];
    }
    return [...out, stage];
  }, []);
}

/** Chapter summaries as the prompt sees them. Numbers only — no prose to localise. */
export function digestOf(notes: readonly ChapterNote[], points = 3): string {
  return notes
    .map((n) => {
      const lines = n.keyPoints.slice(0, points).map((k) => `    - ${k}`).join('\n');
      return `[${n.idx}] ${n.title}\n  ${n.gist}${lines ? `\n${lines}` : ''}`;
    })
    .join('\n\n');
}
