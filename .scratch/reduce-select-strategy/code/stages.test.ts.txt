import { describe, expect, test } from 'bun:test';
import { repairStages } from '../../src/pipeline/stages';

const fallback = (n: number) => `阶段${n}`;
const ids = ['s0', 's1', 's2', 's3'];

describe('repairStages', () => {
  test('按模型给的顺序，每个 id 恰好一次', () => {
    const s = repairStages([{ title: '甲', ids: ['s2', 's0'] }, { title: '乙', ids: ['s1', 's3'] }], ids, fallback);
    expect(s).toEqual([{ title: '甲', ids: ['s2', 's0'] }, { title: '乙', ids: ['s1', 's3'] }]);
  });

  test('模型漏掉的站并进最后一个阶段，重复和未知的 id 被丢弃', () => {
    const s = repairStages([{ title: '甲', ids: ['s0', 's0', 'x9', 's1'] }], ids, fallback);
    expect(s.flatMap((x) => x.ids)).toEqual(['s0', 's1', 's2', 's3']);
  });

  test('只有一站的阶段并进相邻阶段', () => {
    const s = repairStages([{ title: '甲', ids: ['s0'] }, { title: '乙', ids: ['s1', 's2', 's3'] }], ids, fallback);
    expect(s).toEqual([{ title: '乙', ids: ['s0', 's1', 's2', 's3'] }]);
  });

  test('模型什么都没给时用兜底名装下全部站', () => {
    expect(repairStages([], ids, fallback)).toEqual([{ title: '阶段1', ids }]);
  });
});
